```python
class StrategyValidationPipeline:
    """
    策略验证流水线 - 防止过拟合的核心工具
    """
    def __init__(self):
        self.validation_methods = [
            ('time_split', TimeSplitValidation()),
            ('walk_forward', WalkForwardValidation()),
            ('monte_carlo', MonteCarloValidation()),
            ('regime_aware', RegimeAwareValidation())
        ]
    
    def validate_strategy(self, strategy, full_data):
        """
        执行多层验证，返回稳健性评分
        """
        validation_results = {}
        
        # 1. 基础时间分割验证
        train_data, test_data = time_split(full_data, ratio=0.6)
        base_result = self._train_and_test(strategy, train_data, test_data)
        validation_results['base'] = base_result
        
        # 2. 前向遍历验证（检测参数稳定性）
        wf_results = []
        for train_window, test_window in walk_forward_windows(full_data):
            result = self._train_and_test(strategy, train_window, test_window)
            wf_results.append(result)
        validation_results['walk_forward'] = wf_results
        
        # 3. 计算稳健性评分（0-100）
        robustness_score = self._calculate_robustness_score(validation_results)
        
        return {
            'score': robustness_score,
            'details': validation_results,
            'recommendation': '通过' if robustness_score > 70 else '拒绝'
        }
    
    def _calculate_robustness_score(self, results):
        """
        加权计算策略稳健性评分
        权重分配：
        - 样本外表现一致性: 40%
        - 参数敏感性: 30%
        - 市场状态适应性: 30%
        """
        score = 0
        
        # 检查样本外表现一致性
        out_of_sample_performance = []
        for method, result in results.items():
            if hasattr(result, 'sharpe'):
                out_of_sample_performance.append(result.sharpe)
        
        # 计算变异系数（越低越好）
        if len(out_of_sample_performance) > 1:
            cv = np.std(out_of_sample_performance) / np.mean(out_of_sample_performance)
            consistency_score = max(0, 100 * (1 - cv))
            score += consistency_score * 0.4
        
        # 参数敏感性评分（略）
        # 市场适应性评分（略）
        
        return min(100, score)
```

```python
class OverfittingDetector:
    """
    检测策略是否过拟合的多种方法
    """
    def __init__(self):
        self.detectors = {
            'train_test_gap': self.detect_train_test_gap,
            'parameter_sensitivity': self.detect_parameter_sensitivity,
            'feature_importance_stability': self.detect_feature_instability
        }
    
    def detect_train_test_gap(self, train_performance, test_performance):
        """
        检测训练集与测试集表现差距
        """
        metrics_to_compare = ['sharpe', 'max_drawdown', 'win_rate']
        
        gaps = {}
        for metric in metrics_to_compare:
            train_val = train_performance.get(metric, 0)
            test_val = test_performance.get(metric, 0)
            
            if train_val != 0:
                gap_pct = abs(train_val - test_val) / abs(train_val)
                gaps[metric] = gap_pct
        
        # 如果有任一指标差距超过50%，标记为可疑
        is_suspicious = any(gap > 0.5 for gap in gaps.values())
        
        return {
            'is_suspicious': is_suspicious,
            'gaps': gaps,
            'score': 100 * (1 - max(gaps.values()) if gaps else 1)
        }
    
    def generate_overfitting_report(self, strategy, data):
        """
        生成完整的过拟合检测报告
        """
        report = {}
        
        # 执行所有检测方法
        for detector_name, detector_func in self.detectors.items():
            detection_result = detector_func(strategy, data)
            report[detector_name] = detection_result
        
        # 计算综合过拟合风险评分
        overall_risk = self._calculate_overall_risk(report)
        
        report['overall'] = {
            'risk_score': overall_risk,
            'risk_level': self._risk_level(overall_risk),
            'recommendations': self._generate_recommendations(report)
        }
        
        return report
```

```python
class MarketRegimeClassifier:
    """
    基于规则+轻量ML的市场状态分类器
    """
    def __init__(self, config=None):
        self.config = config or {
            'lookback_window': 60,  # 60个交易日
            'regimes': ['trend_up', 'trend_down', 'range_high_vol', 'range_low_vol']
        }
        
        # 特征计算器
        self.feature_extractor = MarketFeatureExtractor()
        
        # 简单的规则分类器（避免复杂ML引入的不稳定性）
        self.rule_based_classifier = RuleBasedClassifier()
        
    def classify_current_market(self, price_data):
        """
        分类当前市场状态
        """
        # 提取特征
        features = self.feature_extractor.extract(price_data)
        
        # 使用规则分类
        regime = self.rule_based_classifier.classify(features)
        
        # 添加置信度计算
        confidence = self._calculate_confidence(features, regime)
        
        return {
            'regime': regime,
            'confidence': confidence,
            'features': features,
            'timestamp': price_data.index[-1]
        }
    
    def _calculate_confidence(self, features, regime):
        """
        计算分类置信度（基于特征与典型值的距离）
        """
        # 获取该市场状态的典型特征值
        typical_features = self.regime_prototypes.get(regime, {})
        
        if not typical_features:
            return 0.7  # 默认置信度
        
        # 计算特征相似度
        similarity_scores = []
        for feature_name in ['trend_strength', 'volatility', 'volume_ratio']:
            if feature_name in features and feature_name in typical_features:
                actual = features[feature_name]
                typical = typical_features[feature_name]
                # 计算归一化距离
                distance = abs(actual - typical) / (abs(typical) + 1e-8)
                similarity = max(0, 1 - distance)
                similarity_scores.append(similarity)
        
        return np.mean(similarity_scores) if similarity_scores else 0.5


class RuleBasedClassifier:
    """
    基于规则的市场状态分类
    """
    def classify(self, features):
        trend_strength = features.get('trend_strength', 0)
        volatility = features.get('volatility', 0)
        atr_ratio = features.get('atr_ratio', 1)
        
        # 简单规则分类
        if trend_strength > 0.3:
            return 'trend_up'
        elif trend_strength < -0.3:
            return 'trend_down'
        elif volatility > 0.02 and atr_ratio > 1.2:
            return 'range_high_vol'
        else:
            return 'range_low_vol'
```

```python
class FactorEffectivenessScorer:
    """
    从多个维度评估因子有效性
    """
    def __init__(self):
        self.metrics_weights = {
            'ic_mean': 0.25,        # 信息系数均值
            'ic_ir': 0.20,         # 信息比率
            'ic_std': -0.15,       # IC标准差（负权重）
            'turnover': -0.15,     # 换手率（负权重）
            'decay_half_life': 0.15, # 半衰期
            'regime_stability': 0.10 # 市场状态稳定性
        }
    
    def score_factor(self, factor_series, returns_series, regime_series=None):
        """
        计算因子综合评分（0-100）
        """
        scores = {}
        
        # 1. 计算信息系数相关指标
        ic_series = calculate_ic_series(factor_series, returns_series)
        scores['ic_mean'] = self._normalize_score(np.mean(ic_series), -0.1, 0.1)
        scores['ic_ir'] = self._normalize_score(
            np.mean(ic_series) / (np.std(ic_series) + 1e-8), 
            0, 2
        )
        scores['ic_std'] = self._normalize_score(np.std(ic_series), 0, 0.05, invert=True)
        
        # 2. 计算换手率
        turnover = calculate_factor_turnover(factor_series)
        scores['turnover'] = self._normalize_score(turnover, 0, 0.5, invert=True)
        
        # 3. 计算衰减半衰期
        decay_rate = calculate_decay_rate(ic_series)
        half_life = -np.log(2) / (decay_rate + 1e-8)
        scores['decay_half_life'] = self._normalize_score(half_life, 5, 30)
        
        # 4. 市场状态稳定性
        if regime_series is not None:
            stability_score = self._calculate_regime_stability(
                ic_series, regime_series
            )
            scores['regime_stability'] = stability_score
        
        # 5. 加权计算总分
        total_score = 0
        for metric, weight in self.metrics_weights.items():
            if metric in scores:
                total_score += scores[metric] * weight
        
        # 转换为0-100分
        final_score = max(0, min(100, 50 + total_score * 50))
        
        return {
            'score': final_score,
            'details': scores,
            'grade': self._score_to_grade(final_score)
        }
    
    def select_top_factors(self, factor_dict, returns_series, top_n=10):
        """
        选择得分最高的因子
        """
        factor_scores = {}
        
        for factor_name, factor_data in factor_dict.items():
            score_result = self.score_factor(factor_data, returns_series)
            factor_scores[factor_name] = score_result['score']
        
        # 按得分排序
        sorted_factors = sorted(
            factor_scores.items(), 
            key=lambda x: x[1], 
            reverse=True
        )
        
        return {
            'top_factors': sorted_factors[:top_n],
            'all_scores': factor_scores
        }
```

```python
class StrategyPortfolioOptimizer:
    """
    基于风险和相关性构建策略组合
    """
    def __init__(self, optimization_method='risk_parity'):
        self.method = optimization_method
        self.min_correlation_threshold = 0.3
        
    def optimize_portfolio(self, strategies_returns, constraints=None):
        """
        优化策略组合权重
        """
        # 1. 计算策略间相关性
        correlation_matrix = strategies_returns.corr()
        
        # 2. 筛选低相关性策略组合
        selected_strategies = self._select_diverse_strategies(
            correlation_matrix, 
            strategies_returns.columns.tolist()
        )
        
        # 3. 根据优化方法计算权重
        if self.method == 'risk_parity':
            weights = self._risk_parity_optimization(
                strategies_returns[selected_strategies]
            )
        elif self.method == 'equal_risk_contribution':
            weights = self._equal_risk_contribution(
                strategies_returns[selected_strategies]
            )
        else:  # 等权重
            n = len(selected_strategies)
            weights = {s: 1.0/n for s in selected_strategies}
        
        # 4. 应用约束
        if constraints:
            weights = self._apply_constraints(weights, constraints)
        
        return {
            'weights': weights,
            'selected_strategies': selected_strategies,
            'correlation_matrix': correlation_matrix,
            'expected_sharpe': self._estimate_sharpe(strategies_returns, weights)
        }
    
    def _select_diverse_strategies(self, corr_matrix, all_strategies, max_strategies=8):
        """
        选择低相关性的策略组合
        """
        selected = []
        remaining = all_strategies.copy()
        
        # 先选择夏普比率最高的策略
        # 这里简化处理，实际应从外部传入策略性能数据
        if len(all_strategies) > 0:
            selected.append(all_strategies[0])
            remaining.remove(all_strategies[0])
        
        # 贪心选择：每次选择与已选策略平均相关性最低的
        while len(selected) < min(max_strategies, len(all_strategies)) and remaining:
            best_candidate = None
            best_avg_correlation = float('inf')
            
            for candidate in remaining:
                # 计算候选策略与已选策略的平均相关性
                avg_corr = np.mean([
                    abs(corr_matrix.loc[candidate, s]) 
                    for s in selected
                ])
                
                if avg_corr < best_avg_correlation:
                    best_avg_correlation = avg_corr
                    best_candidate = candidate
            
            # 如果平均相关性过高，停止添加
            if best_avg_correlation > 0.6:
                break
            
            selected.append(best_candidate)
            remaining.remove(best_candidate)
        
        return selected
```

```python
            # 执行验证
            step_result = validation_func(strategy, historical_data)
            results[step_name] = step_result
            
            # 如果验证失败，停止流程
            if not step_result['passed']:
                print(f"验证失败: {step_result['reason']}")
                return {
                    'passed': False,
                    'failed_step': step_name,
                    'results': results
                }
            
            # 根据验证结果调整策略
            if step_result.get('adjustments'):
                strategy = self._apply_adjustments(strategy, step_result['adjustments'])
        
        return {
            'passed': True,
            'results': results,
            'final_strategy': strategy,
            'recommended_initial_capital': self._calculate_initial_capital(strategy)
        }
    
    def validate_paper_trading(self, strategy, historical_data):
        """
        模拟交易验证
        """
        # 使用最近的3个月数据进行模拟交易
        recent_data = historical_data.iloc[-90:]
        
        # 执行模拟交易
        paper_results = execute_paper_trading(strategy, recent_data)
        
        # 评估模拟交易结果
        evaluation = evaluate_paper_results(paper_results)
        
        # 检查关键指标
        passed = all([
            evaluation['sharpe'] > 1.0,
            evaluation['max_drawdown'] < 0.15,
            evaluation['win_rate'] > 0.45
        ])
        
        return {
            'passed': passed,
            'paper_results': paper_results,
            'evaluation': evaluation,
            'adjustments': self._suggest_adjustments(evaluation) if not passed else None
        }
```

## 八、总结

本计划的核心思想是：**先验证，后优化；先有效，后高效**。我们放弃追求功能的完备性，转而追求策略的有效性和稳健性。

### 关键原则：
1. **实证优先**：所有决策基于数据和验证结果
2. **渐进改进**：从小而简单的功能开始，逐步扩展
3. **风险控制**：始终把防止过拟合和策略失效放在首位
4. **实用导向**：每个功能都要解决实际策略开发中的问题

通过12周的系统性开发，我们将建立一个能够持续产生稳健盈利策略的量化系统，为实盘交易打下坚实基础。
